  <script>

    const $ = (id) => document.getElementById(id);


    function fmt(n) {

      return new Intl.NumberFormat('zh-CN', { maximumFractionDigits: 2 }).format(n);

    }


    function calc() {

      const base = Number($('base').value) || 0;

      const included = Number($('included').value) || 0;

      const overage = Number($('overage').value) || 0;

      const pages = Number($('pages').value) || 0;

      const months = Number($('months').value) || 0;

      const once = Number($('once').value) || 0;


      const extraPages = Math.max(0, pages - included);

      const monthly = base + extraPages * overage;

      $('monthly').textContent = fmt(monthly) + ' 元';


      if (months > 0) {

        const total = monthly * months + once;

        $('total').textContent =

          `租期 ${months} 个月 + 一次性费用：合计约 ${fmt(total)} 元（月均 ${fmt(total / months)} 元）`;

      } else {

        $('total').textContent = once > 0 ? `含一次性费用 ${fmt(once)} 元（未乘租期）` : '';

      }

    }


    ['base', 'included', 'overage', 'pages', 'months', 'once'].forEach((id) => {

      $(id).addEventListener('input', calc);

    });

    calc();

  </script>
